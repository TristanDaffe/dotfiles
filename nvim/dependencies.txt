# Bootstrap dependencies (install before first launch)
#
# Language servers and DAP adapters are listed in plugin/50-lsp.lua (MASON) and
# parsers in plugin/30-treesitter.lua; both install on first launch. Only what Mason
# cannot provide is listed here.

## core
brew install ripgrep     # Telescope live_grep
brew install fd          # Telescope find_files
brew install node        # Copilot and the node-based language servers
brew install tree-sitter-cli  # nvim-treesitter (main) builds parsers with it, needs >= 0.26.1
xcode-select --install   # C compiler for those parsers

## java
brew install openjdk@21  # runs eclipse.jdt.ls; it refuses to start on older
brew install maven
brew install gradle
# python3 (macOS ships one) runs Mason's jdtls wrapper
#
# Project JDKs are looked up in lua/lang/java.lua under `runtimes`; missing ones
# are skipped:
#   JavaSE-21   /opt/homebrew/opt/openjdk@21
#   JavaSE-17   /opt/homebrew/opt/openjdk@17
#   JavaSE-1.8  ~/Library/Java/JavaVirtualMachines/corretto-1.8*
#
# Lombok ships with Mason's jdtls package; lua/lang/java.lua wires it in.

## proto
# protols finds the well-known types (google/protobuf/*.proto) through
# pkg-config. Without `brew install protobuf` those imports stay unresolved;
# everything else works.

## web
# GraphQL completion reads the schema named in the project's graphql.config.*,
# so a project pointing at a local backend needs that backend running.

## ansible
brew install ansible       # ansible-doc (K) and ansible-playbook (<leader>ar)
brew install ansible-lint  # ansiblels lint diagnostics; without it only syntax is checked
