# Bootstrap dependencies (install before first launch)

# macOS:
brew install ripgrep   # Telescope live_grep
brew install fd        # Telescope find_files
brew install node      # Copilot and most LSPs (ts_ls, angularls, html, cssls, yamlls, bashls, ansiblels)

# vscode-html-language-server, vscode-css-language-server) is installed
# automatically by Mason on first launch via `ensure_installed`.
#
